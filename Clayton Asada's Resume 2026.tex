% One text box, with the original 6 pt left inset and no table-cell padding.
\newcommand{\cvtext}[1]{%
  \par\noindent\hspace*{\cvinset}%
  \parbox[t]{\dimexpr\linewidth-\cvinset\relax}{#1}\par
}

\newcommand{\cvsectionheading}[2]{%
  \par\addvspace{\cvsectionbefore}%
  \Needspace{4\baselineskip}%
  \nointerlineskip
  \cvtext{{#1\bfseries\color{darkcol}\MakeUppercase{#2}\par}%
    \vspace{3pt}\nointerlineskip
    \hbox{\color{maincol}\rule{\cvunderlinewidth}{2pt}}}%
  \nobreak\vspace{\cvsectionafter}%
}
\newcommand{\cvsection}[1]{\cvsectionheading{\LARGE}{#1}}
\newcommand{\cvleftsection}[1]{%
  \begingroup
  \setlength{\cvsectionbefore}{12pt}%
  \setlength{\cvsectionafter}{7pt}%
  \cvsectionheading{\large}{#1}%
  \endgroup
}

% Sidebar skill rows: the name can wrap, and the duration stays right-aligned.
\newcommand{\cvother}[2]{%
  \par\noindent
  \begin{tabular}{@{\hspace{\cvinset}}%
    >{\raggedright\arraybackslash}p{\dimexpr\linewidth-3\cvinset-30pt\relax}%
    @{\hspace{\cvinset}}>{\raggedleft\arraybackslash}p{30pt}%
    @{\hspace{\cvinset}}}
    \textbf{#1} & \textcolor{maincol}{#2}
  \end{tabular}\par
}

\newcommand{\cvlist}[1]{%
  \begin{itemize}[leftmargin=24pt,rightmargin=0pt,topsep=4pt,
    itemsep=2pt,parsep=0pt,partopsep=0pt]
    #1
  \end{itemize}%
}

% A heading is indivisible, but the rest of a job may break between blocks.
% #1 is the badge-width fraction; #2 date; #3 title; #4 employer.
\newcommand{\cvjobheading}[4]{%
  \begingroup
  \setlength{\cvheadingwidth}{\dimexpr\linewidth-\cvinset-12pt\relax}%
  \setlength{\cvdatewidth}{#1\linewidth}%
  \noindent\hspace*{\cvinset}%
  \begin{tabular}{@{}>{\raggedright\arraybackslash}%
    p{\dimexpr\cvheadingwidth-\cvdatewidth-8pt\relax}@{\hspace{8pt}}r@{}}
    \textbf{#3} &
    \colorbox{maincol}{\makebox[\dimexpr\cvdatewidth-2\fboxsep\relax]{%
      \textcolor{white}{#2}}}\\
    \textcolor{maincol}{\textbf{#4}} &
  \end{tabular}\par
  \endgroup
}

% Inline technology rows cannot float away from the corresponding job.
% Each logo fits the same box without stretching its original aspect ratio.
\newcommand{\cvtech}[2]{%
  \mbox{\makebox[\cviconsize]{%
    \includegraphics[width=\cviconsize,height=\cviconsize,keepaspectratio]{icons/#1}}%
    \hspace{2pt}#2}%
}
\newcommand{\cvtechrow}[1]{%
  \cvtext{\begin{tabular*}{\linewidth}{@{}l@{\extracolsep{\fill}}llll@{}}
    #1
  \end{tabular*}}%
}
\newcommand{\cvsubheading}[1]{%
  \par\addvspace{\cvsubsectionsep}%
  \Needspace{3\baselineskip}%
  \cvtext{\textbf{#1}}\nobreak\vspace{4pt}%
}

% Named job fields avoid long sequences of empty positional arguments.
\makeatletter
\define@key{cvevent}{date}{\def\cv@date{#1}}
\define@key{cvevent}{title}{\def\cv@title{#1}}
\define@key{cvevent}{employer}{\def\cv@employer{#1}}
\define@key{cvevent}{description}{\def\cv@description{#1}}
\define@key{cvevent}{work}{\def\cv@work{#1}}
\define@key{cvevent}{technologies}{\def\cv@technologies{#1}}
\define@key{cvevent}{achievements}{\def\cv@achievements{#1}}
\newcommand{\cvevent}[1]{%
  \par\begingroup
  \setkeys{cvevent}{date={},title={},employer={},description={},work={},%
    technologies={},achievements={}}%
  \setkeys{cvevent}{#1}%
  \Needspace{6\baselineskip}%
  \cvjobheading{0.35}{\cv@date}{\cv@title}{\cv@employer}%
  \nobreak
  \ifdefempty{\cv@description}{}{%
    \vspace{5pt}\cvtext{\cv@description}}%
  \ifdefempty{\cv@work}{}{\cv@work}%
  \ifdefempty{\cv@technologies}{}{%
    \cvsubheading{Technology Stack:}\cvtechrow{\cv@technologies}}%
  \ifdefempty{\cv@achievements}{}{%
    \cvsubheading{Achievements:}\cv@achievements}%
  \par\endgroup\addvspace{\cventrysep}%
}
\makeatother
